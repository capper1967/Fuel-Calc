  <label>Litres per Fill:</label>
  <input type="number" id="litres" step="0.1">

  <label>KMs per Tank:</label>
  <input type="number" id="kmsPerTank" step="1">

  <label>Trip Distance (KM):</label>
  <input type="number" id="tripKms" step="1">

  <button onclick="calculate()">Calculate</button>

  <div class="result" id="output">
    -
  </div>

<script>
function calculate() {
  const price = parseFloat(document.getElementById('price').value);
  const litres = parseFloat(document.getElementById('litres').value);
  const kmsPerTank = parseFloat(document.getElementById('kmsPerTank').value);
  const tripKms = parseFloat(document.getElementById('tripKms').value);

  if (!price || !litres || !kmsPerTank || !tripKms) {
    alert("Please fill in all fields");
    return;
  }

  // Cost per tank
  const costPerTank = price * litres;

  // Cost per km
  const costPerKm = costPerTank / kmsPerTank;

  // Trip cost
  const tripCost = costPerKm * tripKms;

  document.getElementById('output').innerHTML = `
    <strong>Cost per KM:</strong> $${costPerKm.toFixed(3)}<br>
    <strong>Trip Cost:</strong> $${tripCost.toFixed(2)}
  `;
}

// Register Service Worker
if ('serviceWorker' in navigator) {
  navigator.serviceWorker.register('sw.js');
}
</script>
